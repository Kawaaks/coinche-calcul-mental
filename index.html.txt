<script>
const suits = [
  {name:"Trèfle", symbol:"♣", red:false},
  {name:"Carreau", symbol:"♦", red:true},
  {name:"Cœur", symbol:"♥", red:true},
  {name:"Pique", symbol:"♠", red:false}
];
const ranks = ["7","8","9","10","V","D","R","A"];
const classicTrump = {"7":0,"8":0,"9":14,"10":10,"V":20,"D":3,"R":4,"A":11};
const classicPlain = {"7":0,"8":0,"9":0,"10":10,"V":2,"D":3,"R":4,"A":11};
const noTrump = {"7":0,"8":0,"9":0,"10":10,"V":2,"D":3,"R":4,"A":19};
const allTrump = {"7":0,"8":0,"9":14,"10":10,"V":20,"D":3,"R":4,"A":11};

let selectedMode="classic", deck=[], index=0, cumulative=0, correct=0, errors=0, streak=0, trump=null;
let startTime=null,endTime=null;
let valuesVisible = localStorage.getItem("coincheValuesVisible") !== "false";
let locked=false;

const $ = id => document.getElementById(id);
document.querySelectorAll(".mode").forEach(btn=>{
  btn.addEventListener("click",()=>{
    document.querySelectorAll(".mode").forEach(b=>b.classList.remove("active"));
    btn.classList.add("active"); selectedMode=btn.dataset.mode;
    const texts={
      classic:"Une couleur d’atout est tirée au hasard. Les cartes valent 152 points, puis le dix de der porte le total à 162.",
      all:"Toutes les couleurs sont atout. Le paquet vaut 248 points, puis 258 avec le dix de der. Le résultat final est aussi affiché converti sur 162.",
      none:"Aucun atout. Les As valent 19 points : 152 points dans les cartes, puis 162 avec le dix de der."
    };
    $("modeHelp").textContent=texts[selectedMode];
  });
});

function shuffle(a){
  for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]]}
  return a;
}
function makeDeck(){
  trump = selectedMode==="classic" ? suits[Math.floor(Math.random()*4)] : null;
  const d=[];
  suits.forEach(s=>ranks.forEach(r=>{
    let value;
    if(selectedMode==="classic") value=(s===trump?classicTrump:classicPlain)[r];
    else if(selectedMode==="none") value=noTrump[r];
    else value=allTrump[r];
    d.push({suit:s,rank:r,value});
  }));
  shuffle(d);
  d.push({bonus:true,rank:"+10",value:10,suit:{name:"Dix de der",symbol:"★",red:false}});
  return d;
}
function updateValueVisibility(){
  const pill = $("valuePill");
  pill.classList.toggle("hidden-value", !valuesVisible);
  $("valueToggle").setAttribute("aria-pressed", String(valuesVisible));
  $("valueToggle").textContent = valuesVisible ? "👁 Valeurs visibles" : "🙈 Valeurs cachées";
  localStorage.setItem("coincheValuesVisible", String(valuesVisible));
}
function start(){
  deck=makeDeck();index=0;cumulative=0;correct=0;errors=0;streak=0;locked=false;startTime=null;endTime=null;
  $("setup").classList.add("hidden");$("finish").classList.add("hidden");$("game").classList.remove("hidden");
  const labels={classic:`Classique · Atout ${trump.symbol} ${trump.name}`,all:"Tout atout",none:"Sans atout"};
  $("modeLabel").textContent=labels[selectedMode]; updateStats(); showCard();
}
function showCard(){
  if(index>=deck.length){finish();return}
  const c=deck[index], card=$("card");
  card.classList.toggle("red",c.suit.red);
  $("topCorner").textContent=`${c.rank} ${c.suit.symbol}`;
  $("bottomCorner").textContent=`${c.rank} ${c.suit.symbol}`;
  $("bigSuit").textContent=c.suit.symbol;
  $("valuePill").textContent=c.bonus ? "Bonus du dernier pli : 10" : `Valeur : ${c.value}`;
  updateValueVisibility();
  $("progress").textContent=`${index+1} / ${deck.length}`;
  $("answer").value=""; $("answer").focus();
  $("feedback").textContent=""; $("feedback").className="feedback";
}
function validate(){
  if(locked)return;
  const raw=$("answer").value.trim().replace(",",".");
  if(raw==="")return;
  const given=Number(raw), expected=cumulative+deck[index].value;
  if(!Number.isFinite(given))return;
  if(Math.abs(given-expected)<0.001){
    if(startTime===null) startTime=Date.now();
    correct++;streak++;cumulative=expected;
    $("feedback").textContent=`Exact ! Cumul : ${cumulative}`;
    $("feedback").className="feedback ok"; locked=true; updateStats();
    setTimeout(()=>{index++;locked=false;showCard()},500);
  }else{
    errors++;streak=0;
    $("feedback").textContent=`Pas tout à fait. Essaie encore.`;
    $("feedback").className="feedback bad";updateStats();$("answer").select();
  }
}
function updateStats(){$("score").textContent=correct;$("errors").textContent=errors;$("streak").textContent=streak}
function finish(){
  endTime=Date.now();
  $("game").classList.add("hidden");$("finish").classList.remove("hidden");
  const total=selectedMode==="all"?258:162;
  const conversion=selectedMode==="all"?" Après conversion officielle : 162 / 162.":"";
  const elapsed = startTime ? Math.floor((endTime-startTime)/1000) : 0;
  const min = Math.floor(elapsed/60);
  const sec = String(elapsed%60).padStart(2,"0");
  $("finishText").innerHTML=`Total obtenu : <b>${total}</b>.${conversion}<br><b>Temps : ${min}:${sec}</b><br>${correct} bonnes réponses et ${errors} erreur${errors>1?"s":""}.`;
}
$("valueToggle").addEventListener("click",()=>{
  valuesVisible=!valuesVisible;
  updateValueVisibility();
  $("answer").focus();
});
$("startBtn").addEventListener("click",start);
$("submitBtn").addEventListener("click",validate);
$("answer").addEventListener("keydown",e=>{if(e.key==="Enter")validate()});
$("quitBtn").addEventListener("click",()=>{ $("game").classList.add("hidden");$("setup").classList.remove("hidden")});
$("againBtn").addEventListener("click",()=>{ $("finish").classList.add("hidden");$("setup").classList.remove("hidden")});
</script>
